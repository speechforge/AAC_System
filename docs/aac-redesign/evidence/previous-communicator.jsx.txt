                      id="message-type-select" aria-label={sstr('msgType',language)}
                      value={userAct}
                      onChange={(e) => setUserAct(e.target.value)}
                      className="px-2 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 text-xs font-bold cursor-pointer"
                      title={ui("Message type: leave on Auto to let the app decide; choose one to override", language)}
                    >
                      <option value="auto">{sstr('typeAuto', language)}</option>
                      <option value="REQUEST">{sstr('typeRequest', language)}</option>
                      <option value="FEELING">{sstr('typeFeeling', language)}</option>
                      <option value="QUESTION">{sstr('typeQuestion', language)}</option>
                    </select>
                  </label>

                  <button
                    id="polish-button"
                    onClick={handleGenerateSentence}
                    disabled={isGenerating || (tokens.length === 0 && !inputText)}
                    className="support-btn tap-ripple spring-press"
                    aria-label={ui("Polish",language)} title={ui("Polish: suggest a clearer sentence. Nothing is spoken until you choose.",language)}
                  >
                    {isGenerating ? (
                      <>
                        <span className="w-4 h-4 border-2 border-current border-t-transparent rounded-full animate-spin"></span>
                        <span>{ui("Composing Sentence…", language)}</span>
                      </>
                    ) : (
                      <>
                        <Icons.Sparkles />
                        <span className="toolbar-label">{ui("Polish", language)}</span>
                      </>
                    )}
                  </button>
                </div>

              {/* ─── DEDICATED INFINITY-GRAM PREDICTIONS SHELF ─── */}

                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-2">
                    <span className="text-[11px] font-black uppercase tracking-wider text-slate-600 dark:text-slate-300 flex items-center gap-1.5">
                      <span>{ui("⚡ Next Word Predictions", language)}</span>
                      {isPredicting && <span className="w-2 h-2 rounded-full bg-brand-500 animate-ping"></span>}
                    </span>
                  </div>
                  <span className="text-[10px] text-slate-400 dark:text-slate-500 font-semibold">
                    {tokens.length > 0 ? `Context: ${tokens.length} words` : ui("Conversation Starters", language)}
                  </span>
                </div>

                {/* Prediction Pills Ribbon */}
                <div className="flex items-center gap-2 overflow-x-auto no-scrollbar py-1">
                  {predictions.map((pred, i) => (
                    <button
                      key={i}
                      onClick={() => addToken({ word: pred.word, emoji: pred.emoji || '✨' })}
                      className="tap-ripple spring-press flex-shrink-0 inline-flex items-center gap-2 px-3.5 py-2 rounded-xl bg-white hover:bg-brand-50 dark:bg-slate-800 dark:hover:bg-slate-700 text-slate-900 dark:text-white font-extrabold text-xs sm:text-sm border-2 border-slate-200 hover:border-brand-500 dark:border-slate-700 shadow-xs cursor-pointer transition-all"
                      title={pred.order_label ? `Matched ${pred.order_label}` : undefined}
                    >

                      <span>{pred.word}</span>
                      {pred.order >= 4 && (
                        <span className="text-[9px] font-black px-1.5 py-0.5 rounded-full bg-indigo-100 text-indigo-700 dark:bg-indigo-950 dark:text-indigo-300 border border-indigo-200 dark:border-indigo-800">
                          ∞-{pred.order}{ui("g", language)}</span>
                      )}
                      {pred.prob > 0 && (
                        <span className="text-[10px] font-bold text-slate-400 dark:text-slate-500">
                          {Math.round(pred.prob * 100)}%
                        </span>
                      )}
                    </button>
                  ))}
                </div>
              </div>

            </div>

            <div className="output-workspace" hidden={!generatedResult || compositionView!=='result'}>
              <button type="button" className="back-to-message" onClick={()=>setCompositionView('compose')}>← {sstr('backMessage',language)}</button>
            {/* ─── AI OUTPUT CARD (When Generated) ─── */}
            {generatedResult && generatedResult.outcome === 'clarification' && (
              <div id="result-clarification" className="bg-white dark:bg-slate-900 rounded-2xl border-2 border-amber-500/50 p-4 shadow-sm flex flex-col gap-2.5 animate-fade-in">
                <div className="text-sm font-black text-amber-800 dark:text-amber-300">{sstr('clarifyQ', language)}</div>
                <div className="text-lg font-extrabold text-slate-900 dark:text-white">"{(generatedResult.clarification && generatedResult.clarification.literal) || tokens.map(t => t.label).join(' ')}"</div>
                <div className="flex items-center gap-2 flex-wrap">
                  <button id="clarify-request" onClick={() => handleGenerateSentence('REQUEST')} className="tap-ripple spring-press px-4 py-2 rounded-xl bg-emerald-600 hover:bg-emerald-500 text-white text-sm font-extrabold shadow-xs cursor-pointer">{sstr('optRequest', language)}</button>
                  <button id="clarify-feeling" onClick={() => handleGenerateSentence('FEELING')} className="tap-ripple spring-press px-4 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-sm font-extrabold shadow-xs cursor-pointer">{sstr('optFeeling', language)}</button>
                </div>
              </div>
            )}
            {generatedResult && generatedResult.outcome === 'literal_preview' && (
              <div id="result-literal" className="bg-white dark:bg-slate-900 rounded-2xl border-2 border-slate-400/60 p-4 shadow-sm flex flex-col gap-2.5 animate-fade-in">
                <span className="self-start px-2.5 py-0.5 rounded-full text-[10px] font-black bg-slate-200 text-slate-800 dark:bg-slate-800 dark:text-slate-200">{sstr('literalChip', language)}</span>
                <div className="text-xl sm:text-2xl font-black text-slate-900 dark:text-white">"{generatedResult.text}"</div>
                <div className="flex items-center gap-2">
                  <button id="speak-literal" onClick={() => speak(generatedResult.text, language)} className="tap-ripple spring-press flex items-center gap-1.5 px-3 py-1 rounded-lg bg-slate-700 text-white text-xs font-bold shadow-xs cursor-pointer"><Icons.Volume /><span>{sstr('speakLiteral', language)}</span></button>
                </div>
              </div>
            )}
            {generatedResult && generatedResult.outcome === 'failure' && (
              <div id="result-failure" className="bg-white dark:bg-slate-900 rounded-2xl border-2 border-rose-500/50 p-4 shadow-sm animate-fade-in text-sm font-black text-rose-700">{sstr('failChip', language)}</div>
            )}
            {generatedResult && generatedResult.outcome !== 'clarification' && generatedResult.outcome !== 'literal_preview' && generatedResult.outcome !== 'failure' && (
              <div className="bg-white dark:bg-slate-900 rounded-2xl border-2 border-emerald-500/40 p-4 shadow-sm flex flex-col gap-2.5 animate-fade-in">
                <div className="flex items-center justify-between flex-wrap gap-2">
                  <div className="flex items-center gap-2 flex-wrap">
                    <span className="px-2.5 py-0.5 rounded-full text-[10px] font-black bg-emerald-100 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300 flex items-center gap-1">
                      <Icons.Check />
                      <span>{generatedResult.outcome === 'sentence_template' ? sstr('templateChip', language) : sstr('modelChip', language)}</span>
                    </span>
                    {lastAudioB64 && (
                      <span className="text-[10px] font-bold text-indigo-600 dark:text-indigo-400 bg-indigo-50 dark:bg-indigo-950/50 px-2 py-0.5 rounded-full border border-indigo-200 dark:border-indigo-800">
                        {generatedResult.audio_kind === 'indicf5' ? ui("🎙️ IndicF5", language) : generatedResult.audio_kind === 'f5tts' ? '🎙️ F5-TTS' : ui("Speech unavailable", language)}
                      </span>
                    )}
                  </div>
                  <div className="flex items-center gap-1.5">
                    {/* ─── Feature 6: Bookmark Sentences (Quick Picks) ─── */}
                    <button
                      onClick={() => addBookmark(generatedResult.text)}
                      className={`tap-ripple spring-press flex items-center justify-center w-8 h-8 rounded-lg border text-base ${
                        bookmarks.some(b => b.text === generatedResult.text)
                          ? 'bg-amber-100 border-amber-300 text-amber-600 dark:bg-amber-950/50 dark:border-amber-700'
                          : 'bg-white border-slate-200 text-slate-400 hover:text-amber-500 dark:bg-slate-800 dark:border-slate-700'
                      } ${starPopId ? 'star-pop' : ''}`}
                      title={ui("Save to Bookmarks", language)}
                    >
                      {bookmarks.some(b => b.text === generatedResult.text) ? <Icons.StarFilled /> : <Icons.StarOutline />}
                    </button>
                    <button
                      onClick={() => copyToClipboard(generatedResult.text)}
                      className="tap-ripple spring-press flex items-center gap-1 px-2.5 py-1 rounded-lg bg-slate-100 hover:bg-slate-200 dark:bg-slate-800 text-slate-700 dark:text-slate-300 text-xs font-bold border border-slate-200 dark:border-slate-700 shadow-xs cursor-pointer"
                      title={ui("Copy sentence to clipboard", language)}
                    >
                      <Icons.Copy />
                      <span>{ui("Copy", language)}</span>
                    </button>
                    <button
                      id="use-suggestion" onClick={useSuggestion}
                      className="tap-ripple spring-press flex items-center gap-1.5 px-3 py-1 rounded-lg bg-emerald-600 text-white text-xs font-bold shadow-xs hover:bg-emerald-500"
                    >
                      <Icons.Play />
                      <span className="toolbar-label">{ui("Use suggestion & speak", language)}</span>
                    </button>
                  </div>
                </div>

                <div id="result-sentence" className="text-xl sm:text-2xl font-black text-slate-900 dark:text-white">
                  "{generatedResult.text}"
                </div>

                <IntentContext id="result-intent-context" act={generatedResult.context_intent || generatedResult.act} negation={generatedResult.context_negation} words={generatedResult.selected_words} explanation={generatedResult.caregiver_context?.caregiver_explanation} language={language} />
              </div>
            )}

            </div>
            </section>
            <section className="vocabulary-workspace" aria-label={ui("Word categories",language)}>
            {/* ─── 3. CATEGORY NAVIGATION DOCK & CARD SIZING ─── */}
            <div className="category-dock">
              <div className="category-tabs" aria-label={ui("Word categories", language)}>
                {CATEGORIES.map((cat) => {
                  const isActive = activeCategory === cat.id;
                  const catLabel = cat.label[language] || cat.label.en;
                  return (
                    <button
                      key={cat.id}
                      data-category={cat.id}
                      aria-pressed={isActive}
                      onClick={() => {
                        setActiveCategory(cat.id);
                        if (cat.id === 'search') {
                          setTimeout(() => document.getElementById('gridSearchInput')?.focus(), 50);
                        }
                      }}
                      className={`tap-ripple spring-press flex-shrink-0 flex items-center gap-1.5 px-3.5 py-2 rounded-xl font-bold text-xs border transition-all cursor-pointer ${
                        isActive
                          ? 'bg-slate-900 text-white border-slate-900 dark:bg-white dark:text-slate-900 dark:border-white shadow-sm'
                          : 'bg-white dark:bg-slate-900 text-slate-600 dark:text-slate-300 border-slate-200/80 dark:border-slate-800 hover:border-slate-300'
                      }`}
                    >
                      <SymbolVisual item={{k:{core:'MORE',actions:'GO',food:'FOOD',feelings:'HAPPY',people:'ME',places:'HOME',body:'HEART',chat:'NAMASTE',emergency:'HELP_EMERGENCY',search:'LOOK'}[cat.id],e:cat.icon}} style={visualStyle} photos={photoOverrides} />
                      <span className="category-label">{catLabel}</span>
                    </button>
                  );
                })}
              </div>

            </div>

            {/* ─── 4. SYMBOL BOARD GRID ───────────────────────────────── */}
            <div id="symbolBoard" className="scroll-mt-20 bg-white dark:bg-slate-900 rounded-2xl border border-slate-200/80 dark:border-slate-800 p-4 shadow-sm flex flex-col gap-3">
            
              {/* Optional Search Input if Search Category is Active */}
              {activeCategory === 'search' && (
                <div className="relative">
                  <div className="absolute left-3.5 top-1/2 -translate-y-1/2 text-slate-400">
                    <Icons.Search />
                  </div>
                  <input
                    id="gridSearchInput"
                    type="text"
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                    placeholder={ui("Search over 5,000+ emojis (e.g. apple, doctor, happy, bus)...", language)}
                    className="w-full bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl pl-10 pr-4 py-2 text-sm font-semibold text-slate-800 dark:text-slate-100 focus:outline-none focus:border-brand-500"
                  />
                  {searchQuery && (
                    <button
                      onClick={() => setSearchQuery('')}
                      className="absolute right-3.5 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600 font-bold"
                    >
                      ✕
                    </button>
                  )}
                </div>
              )}

              {/* ─── Feature 1: Custom Grid — add-word form (offline local cache) ─── */}
              <div key={activeCategory} className="symbol-grid" data-density={cardSize} data-category={activeCategory}>
                {displayedSymbols.map(item => {
                  const label=item.labels ? (item.labels[language] || item.labels.en || item.k) : (item.label || item.k);
                  const content=<><SymbolVisual item={item} style={visualStyle} photos={photoOverrides} /><span className="card-label">{label}</span></>;
                  return item.isCustom ? (
                    <div className="custom-symbol-cell" key={item.k}>
                      <button className="symbol-card comm-card flex flex-col items-center text-center border" data-symbol-id={item.k} onClick={() => addToken(item)}>{content}</button>
                      <button className="custom-symbol-remove" aria-label={ui('Remove from My Words',language)+': '+label} onClick={() => removeCustomWord(item.k)}>×</button>
                    </div>
                  ) : <button key={item.k} className="symbol-card comm-card flex flex-col items-center text-center border" data-symbol-id={item.k} onClick={() => addToken(item)}>{content}</button>;
                })}
              </div>

              {activeCategory === 'search' && displayedSymbols.length === 0 && searchQuery && (
                <div className="text-center py-12 text-slate-400 font-medium text-sm">{ui("No emojis found matching \"", language)}{searchQuery}".
                </div>
              )}


            </div>
